60inline auto ModulePIMPL::not_set(const type::input_map& in_inputs) const {
61 assert_mod_();
62 not_set_type probs;
63
64 // This is all of the not ready inputs
65 auto in_probs = not_set_guts_(m_inputs_);
66
67 // Now pull out those set by the property type
68 for(const auto& [k, v] : in_inputs)
69 if(in_probs.count(k)) in_probs.erase(k);
70 if(!in_probs.empty()) probs.emplace("Inputs", std::move(in_probs));
71
72 auto submod_probs = not_set_guts_(run_submods());
73 if(!submod_probs.empty())
74 probs.emplace("Submodules", std::move(submod_probs));
75 return probs;
76}

200inline auto ModulePIMPL::run(type::input_map ps) {
201 auto time_now = time_stamp();
202 m_timer_.reset();
203 assert_mod_();
204 if(is_driver() && !m_resolved_)
205 throw std::runtime_error(
206 "A driver must be run through a ModuleManager, i.e., with "
207 "ModuleManager::run_as or as a submodule of a module that was, so "
208 "that its wiring requests can be resolved.");
209
210 // Check the inputs we were just given
211 for(const auto& [k, v] : ps)
212 if(!v.ready()) throw std::runtime_error("Inputs are not ready");
213
214 // Merge with bound and see if we are ready
215 if(!ready(ps)) {
216 // Make a dummy module with this PIMPL so we can print out why it's not
217 // ready.
218 Module dummy(std::make_unique<ModulePIMPL>(*this));
219 throw std::runtime_error(print_not_ready(dummy, ps));
220 }
221
222 lock();
223
224 ps = merge_inputs_(std::move(ps));
225
226 if(!m_cache_ || !is_memoizable()) {
227 auto rv = m_base_->run(ps, active_submods_());
228 m_timer_.record(time_now);
229 return rv;
230 }
231
232 ps = cache_key_(std::move(ps));
233 if(m_cache_->count(ps)) {
234 m_timer_.record(time_now);
235 return m_cache_->uncache(ps);
236 }
237
238 // not there so run
239 auto rv = m_base_->run(ps, active_submods_());
240
241 // cache result
242 m_cache_->cache(ps, std::move(rv));
243 m_timer_.record(time_now);
244 return m_cache_->uncache(ps);
245}

247inline bool ModulePIMPL::operator==(const ModulePIMPL& rhs) const {
248 if(has_module() != rhs.has_module()) return false;
249 if(locked() != rhs.locked()) return false;
250 if(!has_module()) return true;
251 if(m_user_request_ != rhs.m_user_request_) return false;
252
253 if(std::tie(inputs(), submods(), property_types(),
254 python_property_types()) !=
255 std::tie(rhs.inputs(), rhs.submods(), rhs.property_types(),
256 rhs.python_property_types()))
257 return false;
258 return (*m_base_ == *rhs.m_base_);
259}

271inline typename ModulePIMPL::submod_uuid_map ModulePIMPL::submod_uuids() const {
272 submod_uuid_map rv;
273 for(const auto& [k, v] : submods()) {
274 // Prepend the submodule key to each of its submodules' keys
275 // N.B. This essentially scopes the sub submodule keys
276 for(const auto& [sub_k, sub_v] : v.submod_uuids()) {
277 rv.emplace(k + ":" + sub_k, sub_v);
278 }
279 rv.emplace(k, v.uuid());
280 }
281 return rv;

291inline type::input_map ModulePIMPL::cache_key_(type::input_map inputs) const {
292 // TODO: It probably makes sense to create an Input class which tracks this
293 // and allows using submods as inputs
294 // The fingerprints identify the state of everything below each submodule
295 submod_uuid_map fingerprints;
296 for(const auto& [k, v] : run_submods())
297 fingerprints.emplace(
298 k, v.has_module() ? v.value().m_pimpl_->fingerprint() : "");
299
300 std::string submod_key = "__PLUGIN_PLAY__ SUBMOD KEYS __PLUGIN_PLAY__";
301 ModuleInput temp;
302 temp.set_type<submod_uuid_map>();
303 temp.change(std::move(fingerprints));
304 inputs.emplace(submod_key, temp);
305 return inputs;
306}

308inline void ModulePIMPL::lock() {
309 // An overlay refers to its original's inputs, which must not change
310 if(m_overlay_ && !m_locked_) {
311 lock_original_(*m_overlay_->original);
312 sync_overlay_();
313 }
314 for(auto& [k, v] : active_submods_()) v.lock();
315 m_locked_ = true;
316 m_user_request_.lock();
317}
318
319inline void ModulePIMPL::unlock() {
320 if(m_overlay_) {
321 // Copying an input copies the value it refers to
322 type::input_map inputs(m_inputs_);
323 m_inputs_.swap(inputs);
324 m_overlay_.reset();
325 }
326 m_resolved_.reset();
327 m_fingerprint_.reset();
328 m_locked_ = false;
329 m_user_request_.unlock();
330}

375inline ModulePIMPL::module_ptr ModulePIMPL::make_overlay(
376 const module_ptr& original) {
377 const auto& rhs = *original->m_pimpl_;
378 rhs.assert_mod_();
379
380 auto pimpl = std::make_unique<ModulePIMPL>();
381 pimpl->m_memoizable_ = rhs.m_memoizable_;
382 pimpl->m_base_ = rhs.m_base_;
383 pimpl->m_cache_ = rhs.m_cache_;
384 pimpl->m_property_types_ = rhs.m_property_types_;
385 pimpl->m_python_property_types_ = rhs.m_python_property_types_;
386 // An overlay of a driver is a driver, which resolves the same requests
387 pimpl->m_user_request_ = rhs.m_user_request_;
388 pimpl->m_user_request_.unlock();
389 pimpl->m_fingerprints_ = rhs.m_fingerprints_;
390 pimpl->m_overlay_.emplace(OverlayState{original, {}, {}, std::nullopt});
391 pimpl->sync_overlay_();
392
393 auto rv = std::make_shared<Module>(std::move(pimpl));
394 if(original->has_name()) rv->set_name(original->get_name());
395 return rv;
396}

403inline void ModulePIMPL::override_input(const type::key& key, type::any value) {
404 assert_unlocked_overlay_();
405 const auto& inputs = m_overlay_->original->m_pimpl_->m_inputs_;
406 auto itr = inputs.find(key);
407 if(itr == inputs.end())
408 throw std::out_of_range("Module has no input \"" + key + "\"");
409
410 // Keeps the type and bounds checks of the original's input
411 auto input = itr->second.cref_clone();
412 input.change(std::move(value));
413 m_overlay_->input_overrides.insert_or_assign(key, std::move(input));
414 sync_overlay_();
415}

441inline void ModulePIMPL::sync_overlay_() {
442 const auto& rhs = *m_overlay_->original->m_pimpl_;
443
444 // Inputs the overlay does not change refer to the original's values
445 type::input_map inputs;
446 for(const auto& [k, v] : rhs.m_inputs_) inputs.emplace(k, v.cref_clone());
447 for(const auto& [k, v] : m_overlay_->input_overrides)
448 inputs.insert_or_assign(k, v.cref_clone());
449 m_inputs_.swap(inputs);
450
451 auto submods = rhs.m_submods_;
452 for(const auto& [k, v] : m_overlay_->submod_overrides)
453 submods.insert_or_assign(k, v);
454 m_submods_.swap(submods);
455
456 m_memoizable_ = m_overlay_->memoizes.value_or(rhs.m_memoizable_);
457 m_fingerprint_.reset();
458}

460inline void ModulePIMPL::sync_overlays_() const {
461 // N.B. The originals of an unlocked driver's overlays may have changed
462 // since the driver's requests were resolved, e.g., after a failed run.
463 // Syncing keeps the overlays from referring to values that are gone.
464 if(!m_resolved_) return;
465 for(const auto& overlay : m_resolved_->overlays) {
466 auto& pimpl = *overlay->m_pimpl_;
467 if(pimpl.m_locked_) continue;
468 pimpl.sync_overlay_();
469 pimpl.sync_overlays_();
470 }
471}

479inline ModulePIMPL::fingerprint_type ModulePIMPL::fingerprint_(
480 bool& is_final) const {
481 if(m_fingerprint_) {
482 is_final = true;
483 return *m_fingerprint_;
484 }
485 assert_mod_();
486
487 const auto impl = m_base_->has_uuid() ? m_base_->uuid() : type().name();
488 FingerprintTable::Key key{impl, m_memoizable_, {}};
489 is_final = m_locked_;
490 for(const auto& [k, v] : run_submods()) {
491 if(!v.has_module()) {
492 key.submods.emplace(k, "");
493 is_final = false;
494 continue;
495 }
496 bool child_is_final = false;
497 key.submods.emplace(k,
498 v.value().m_pimpl_->fingerprint_(child_is_final));
499 is_final = is_final && child_is_final;
500 }
501
502 if(!m_fingerprints_) m_fingerprints_ = std::make_shared<FingerprintTable>();
503 auto rv = m_fingerprints_->intern(std::move(key), m_inputs_);
504 if(is_final) m_fingerprint_ = rv;
505 return rv;
506}

508inline void ModulePIMPL::lock_original_(Module& original) {
509 // The original may not be ready on its own, e.g., if the overlay binds a
510 // submodule the original leaves unset. Its inputs are what the overlay
511 // refers to, so it is locked, and its submodules are locked if they can be.
512 auto& pimpl = *original.m_pimpl_;
513 if(pimpl.m_locked_) return;
514 if(pimpl.m_overlay_) {
515 lock_original_(*pimpl.m_overlay_->original);
516 pimpl.sync_overlay_();
517 }
518 for(auto& [k, v] : pimpl.active_submods_())
519 if(v.ready()) v.lock();
520 pimpl.m_locked_ = true;
521 pimpl.m_user_request_.lock();
522}
