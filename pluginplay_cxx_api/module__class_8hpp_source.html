868template<typename T>
869void Module::change_input(const type::key& key, T&& value) {
870 assert_not_locked_();
871 if(!inputs().count(key)) {
872 std::string msg = " does not have input \"" + key + "\"";
873 if(has_name())
874 msg = "Module \"" + get_name() + "\"" + msg;
875 else
876 msg = "Unnamed module" + msg;
877 throw std::out_of_range(msg);
878 }
879 get_input_(key).change(std::forward<T>(value));
880}

892template<typename property_type, typename... Args>
893auto Module::run_as(Args&&... args) {
894 check_property_type_(type::rtti{typeid(property_type)});
895 auto temp = inputs();
896 temp = property_type::wrap_inputs(temp, std::forward<Args>(args)...);
897 using r_type = decltype(property_type::unwrap_results(run(temp)));
898 using clean_t = std::decay_t<r_type>;
899 if constexpr(std::is_same_v<clean_t, void>) {
900 property_type::unwrap_results(run(temp));
901 } else {
902 auto rv = property_type::unwrap_results(run(temp));
903
904 if constexpr(std::tuple_size_v<clean_t> == 1) {
905 return std::get<0>(rv);
906 } else {
907 return rv;
908 }
909 }
910}
