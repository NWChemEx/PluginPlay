796template<typename T>
797void Module::change_input(const type::key& key, T&& value) {
798 assert_not_locked_();
799 if(!inputs().count(key)) {
800 std::string msg = " does not have input \"" + key + "\"";
801 if(has_name())
802 msg = "Module \"" + get_name() + "\"" + msg;
803 else
804 msg = "Unnamed module" + msg;
805 throw std::out_of_range(msg);
806 }
807 get_input_(key).change(std::forward<T>(value));
808}

820template<typename property_type, typename... Args>
821auto Module::run_as(Args&&... args) {
822 check_property_type_(type::rtti{typeid(property_type)});
823 auto temp = inputs();
824 temp = property_type::wrap_inputs(temp, std::forward<Args>(args)...);
825 using r_type = decltype(property_type::unwrap_results(run(temp)));
826 using clean_t = std::decay_t<r_type>;
827 if constexpr(std::is_same_v<clean_t, void>) {
828 property_type::unwrap_results(run(temp));
829 } else {
830 auto rv = property_type::unwrap_results(run(temp));
831
832 if constexpr(std::tuple_size_v<clean_t> == 1) {
833 return std::get<0>(rv);
834 } else {
835 return rv;
836 }
837 }
838}

840template<typename ArgsType>

843 check_python_property_type_(pt.name());
