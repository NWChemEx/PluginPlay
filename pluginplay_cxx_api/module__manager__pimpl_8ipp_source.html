41inline void ModuleManagerPIMPL::add_module(type::key key,
42 module_base_ptr base) {
43 if(m_modules.count(key)) throw std::invalid_argument("Key is in use");
44 if(!base)
45 throw std::invalid_argument("Module implementation for key: '" + key +
46 "' is null");
47
48 module_base_ptr stored = base;
49 if(base->is_python()) {
50 // This is a hacky patch to allow multiple python modules to be
51 // added while avoiding the type_index collisions.
52 // TODO: remove when a more elegant solution is determined
53 m_py_bases[key] = base;
54 } else {
55 std::type_index type(base->type());
56 if(!m_bases.count(type)) m_bases[type] = base;
57 stored = m_bases[type];
58 }
59
60 Module mod(std::make_unique<ModulePIMPL>(stored));
61 attach_(key, mod);
62 m_modules.add_module(std::move(key), std::move(mod));
63}

89inline void ModuleManagerPIMPL::load_plugin(plugin_pointer plugin) {
90 if(!plugin) throw std::invalid_argument("Can not load a null plugin");
91
92 const auto& name = plugin->name();
93 if(name.empty())
94 throw std::invalid_argument(
95 "Plugins must have a name. Call set_name in the plugin's "
96 "constructor.");
97 if(has_plugin(name))
98 throw std::invalid_argument("A plugin named '" + name +
99 "' is already loaded");
100
101 // Attach the plugin's modules, then every module reachable from them
102 // through submodules. A plugin may set a submodule and then erase that
103 // module's key; the module is still used, so it needs the runtime and
104 // caches too.
105 std::set<const Module*> visited;
106 std::vector<Module*> to_visit;
107 for(auto& [key, mod] : plugin->m_modules_) {
108 attach_(qualified_key(name, key).to_string(), *mod);
109 visited.insert(mod.get());
110 to_visit.push_back(mod.get());
111 }
112 while(!to_visit.empty()) {
113 auto* mod = to_visit.back();
114 to_visit.pop_back();
115 for(auto& [callback, request] : mod->m_pimpl_->submods()) {
116 if(!request.has_module()) continue;
117 auto& submod = request.value();
118 if(!visited.insert(&submod).second) continue;
119
120 // The module is no longer in the plugin's map, so its cache is
121 // named after the key it was last stored under
122 const auto& last_key =
123 submod.has_name() ? submod.get_name() : callback;
124 attach_(qualified_key(name, last_key).to_string(), submod);
125 to_visit.push_back(&submod);
126 }
127 }
128
129 m_plugins.push_back(std::move(plugin));
130}

132inline void ModuleManagerPIMPL::wire_defaults() {
133 for(const auto& plugin : m_plugins) {
134 auto& modules = plugin->m_modules_;
135 for(const auto& provider_name : plugin->preferred_providers()) {
136 const auto* provider = find_plugin_(provider_name);
137 if(provider == nullptr) continue;
138
139 const auto& provided = provider->m_modules_;
140 for(const auto& record :
141 plugin->default_submods_from(provider_name)) {
142 // The user may have erased the module after loading
143 if(!modules.count(record.module_key)) continue;
144
145 if(!provided.count(record.provider_key)) {
146 if(m_runtime_)
147 m_runtime_->logger().warn(
148 "Plugin '" + plugin->name() + "' asked for module '" +
149 record.provider_key + "' from plugin '" +
150 provider_name + "', which does not have it.");
151 continue;
152 }
153
154 modules.default_submod(record.module_key, record.callback_key,
155 provided.at(record.provider_key));
156 }
157 }
158 }
159}

198inline bool ModuleManagerPIMPL::operator==(
199 const ModuleManagerPIMPL& rhs) const {
200 // Try to get out early
201 if(m_bases.size() != rhs.m_bases.size()) return false;
202 if(m_modules.size() != rhs.m_modules.size()) return false;
203 if(m_plugins.size() != rhs.m_plugins.size()) return false;
204
205 // TODO: Remove with the rest of the python hack
206 if(m_py_bases.size() != rhs.m_py_bases.size()) return false;
207 for(const auto& [k, v] : rhs.m_py_bases) {
208 if(!m_py_bases.count(k)) return false;
209 if(*m_py_bases.at(k) != *v) return false;
210 }
211
212 // Skip checking the values b/c implementations are compared by type
213 for(const auto& [k, v] : rhs.m_bases) {
214 if(!m_bases.count(k)) return false;
215 }
216
217 // Need to check the values b/c user may have switched options
218 if(m_modules != rhs.m_modules) return false;
219
220 // Load order does not matter, so plugins are matched by name
221 for(const auto& plugin : rhs.m_plugins) {
222 const auto* mine = find_plugin_(plugin->name());
223 if(mine == nullptr || *mine != *plugin) return false;
224 }
225
226 return true;
227}

257inline ModuleManagerPIMPL::name_type ModuleManagerPIMPL::resolve_(
258 const qualified_key& key) const {
259 const std::string no_module =
260 "ModuleManager has no module with key: '" + key.to_string() + "'";
261
262 if(key.is_qualified()) {
263 const auto* map = find_map_(key.plugin());
264 if(map == nullptr)
265 throw std::out_of_range("ModuleManager has no plugin named: '" +
266 key.plugin() + "'");
267 if(!map->count(key.key())) throw std::out_of_range(no_module);
268 return key.plugin();
269 }
270
271 name_container_type holders;
272 for(const auto& name : plugin_names())
273 if(find_map_(name)->count(key.key())) holders.push_back(name);
274
275 if(holders.empty()) throw std::out_of_range(no_module);
276 if(holders.size() > 1) {
277 std::string msg = "Module key '" + key.key() +
278 "' is ambiguous. It is used by the plugins:";
279 for(const auto& name : holders)
280 msg += name.empty() ? " '' (the ModuleManager's own modules)" :
281 " '" + name + "'";
282 msg += ". Pass the plugin name too, e.g., {\"" + holders.back() +
283 "\", \"" + key.key() + "\"}.";
284 throw std::invalid_argument(msg);
285 }
286 return holders.front();
287}
