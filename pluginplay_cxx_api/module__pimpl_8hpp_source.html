45inline auto time_stamp() {
46 using namespace std::chrono;
47 const auto now = system_clock::now();
48 const auto now_tt = system_clock::to_time_t(now);
49 const auto ms = duration_cast<milliseconds>(now.time_since_epoch()) % 1000;
50 std::stringstream ss;
51 ss << std::put_time(std::localtime(&now_tt), "%d-%m-%Y %H:%M:%S") << '.'
52 << std::setfill('0') << std::setw(3) << ms.count();
53 return ss.str();
54}
