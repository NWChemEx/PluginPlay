41inline auto time_stamp() {
42 using namespace std::chrono;
43 const auto now = system_clock::now();
44 const auto now_tt = system_clock::to_time_t(now);
45 const auto ms = duration_cast<milliseconds>(now.time_since_epoch()) % 1000;
46 std::stringstream ss;
47 ss << std::put_time(std::localtime(&now_tt), "%d-%m-%Y %H:%M:%S") << '.'
48 << std::setfill('0') << std::setw(3) << ms.count();
49 return ss.str();
50}
