663inline bool ModuleBase::operator==(const ModuleBase& rhs) const noexcept {
664 if(type() != rhs.type()) return false;
665 if(has_description() != rhs.has_description()) return false;
666 if(has_description() && get_desc() != rhs.get_desc()) return false;
667 if(m_citations_ != rhs.m_citations_) return false;
668 if(m_is_python_ != rhs.m_is_python_) return false;
669 return std::tie(inputs(), results(), submods(), property_types(),
670 python_property_types()) ==
671 std::tie(rhs.inputs(), rhs.results(), rhs.submods(),
672 rhs.property_types(), rhs.python_property_types());
673}
