625inline bool ModuleBase::operator==(const ModuleBase& rhs) const noexcept {
626 if(type() != rhs.type()) return false;
627 if(has_description() != rhs.has_description()) return false;
628 if(has_description() && get_desc() != rhs.get_desc()) return false;
629 if(m_citations_ != rhs.m_citations_) return false;
630 if(m_is_python_ != rhs.m_is_python_) return false;
631 return std::tie(inputs(), results(), submods(), property_types(),
632 python_property_types()) ==
633 std::tie(rhs.inputs(), rhs.results(), rhs.submods(),
634 rhs.property_types(), rhs.python_property_types());
635}
