42class FingerprintTable {
43public:
45 using fingerprint_type = std::string;

48 using submod_map_type = std::map<type::key, fingerprint_type>;

51 struct Key {
53 std::string impl;
54
56 bool memoizes;
57
59 submod_map_type submods;
60
62 bool operator<(const Key& rhs) const {
63 return std::tie(impl, memoizes, submods) <
64 std::tie(rhs.impl, rhs.memoizes, rhs.submods);
65 }
66 };

78 fingerprint_type intern(Key key, const type::input_map& inputs) {
79 std::lock_guard<std::mutex> guard(m_mutex_);
80 auto& bucket = m_table_[std::move(key)];
81 for(const auto& [bucket_inputs, id] : bucket)
82 if(bucket_inputs == inputs) return id;
83 // N.B. copying the inputs deep copies their values, so the table does
84 // not refer to the module's values
85 bucket.emplace_back(inputs, utility::generate_uuid());
86 return bucket.back().second;
87 }
88
89private:
91 using bucket_type =
92 std::vector<std::pair<type::input_map, fingerprint_type>>;
93
95 std::mutex m_mutex_;
96
98 std::map<Key, bucket_type> m_table_;
99};
