117template<typename ArgsType>
118type::input_map PythonOnlyPropertyType::wrap_inputs(type::input_map rv,
119 ArgsType&& args) const {
120 if(static_cast<std::size_t>(args.size()) != m_input_order_.size())
121 throw std::runtime_error("PythonOnlyPropertyType \"" + m_name_ +
122 "\": wrong number of positional inputs");
123 for(std::size_t i = 0; i < m_input_order_.size(); ++i)
124 rv.at(m_input_order_[i]).change(PythonWrapper(args[i]));
125 return rv;
126}
